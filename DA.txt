# ============================================================
# 1. IMPORT LIBRARIES
# ============================================================

import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt

# Import train-test split
from sklearn.model_selection import train_test_split

# Import missing value handling
from sklearn.impute import SimpleImputer

# Import categorical data encoding
from sklearn.preprocessing import OneHotEncoder

# Import preprocessing tools
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline

# Import Decision Tree
from sklearn.tree import DecisionTreeClassifier, plot_tree

# Import evaluation metrics
from sklearn.metrics import (
    accuracy_score,
    confusion_matrix,
    classification_report,
    ConfusionMatrixDisplay
)


# ============================================================
# 2. LOAD DATASET
# ============================================================

# Load built-in Titanic dataset
df = sns.load_dataset("titanic")

# Display first 5 rows
df.head()


# ============================================================
# 3. DATASET INFORMATION
# ============================================================

# Display dataset shape
print("Dataset shape:", df.shape)

# Display column names
print("\nColumn names:")
print(df.columns.tolist())

# Display data types
print("\nData types:")
print(df.dtypes)

# Display missing values
print("\nMissing values:")
print(df.isnull().sum())

# Count duplicate rows
print("\nDuplicate rows:", df.duplicated().sum())

# Display statistical summary
print("\nStatistical summary:")
print(df.describe())


# ============================================================
# 4. TARGET VARIABLE ANALYSIS
# ============================================================

# Count survived and not survived passengers
print("\nSurvival counts:")
print(df["survived"].value_counts())

# Calculate survival percentages
print("\nSurvival percentages:")
print(df["survived"].value_counts(normalize=True) * 100)


# ============================================================
# 5. SELECT FEATURES AND TARGET
# ============================================================

# Select independent variables (input features)
features = [
    "pclass",
    "sex",
    "age",
    "sibsp",
    "parch",
    "fare",
    "embarked"
]

X = df[features]

# Select dependent variable (target)
y = df["survived"]


# ============================================================
# 6. IDENTIFY NUMERICAL AND CATEGORICAL FEATURES
# ============================================================

# Numerical columns
numeric_features = [
    "pclass",
    "age",
    "sibsp",
    "parch",
    "fare"
]

# Categorical columns
categorical_features = [
    "sex",
    "embarked"
]

print("Numerical features:", numeric_features)
print("Categorical features:", categorical_features)


# ============================================================
# 7. SPLIT DATASET INTO TRAINING AND TESTING DATA
# ============================================================

# 80% training data and 20% testing data
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=42,
    stratify=y
)

print("Training data shape:", X_train.shape)
print("Testing data shape:", X_test.shape)


# ============================================================
# 8. CREATE NUMERICAL PIPELINE
# ============================================================

# Replace missing numerical values with median
numeric_pipeline = Pipeline(
    steps=[
        ("imputer", SimpleImputer(strategy="median"))
    ]
)


# ============================================================
# 9. CREATE CATEGORICAL PIPELINE
# ============================================================

# Replace missing categorical values with mode
# Convert categorical values into numerical format
categorical_pipeline = Pipeline(
    steps=[
        ("imputer", SimpleImputer(strategy="most_frequent")),
        ("encoder", OneHotEncoder(handle_unknown="ignore"))
    ]
)


# ============================================================
# 10. CREATE COLUMN TRANSFORMER
# ============================================================

# Apply numerical and categorical pipelines separately
preprocessor = ColumnTransformer(
    transformers=[
        ("numeric", numeric_pipeline, numeric_features),
        ("categorical", categorical_pipeline, categorical_features)
    ]
)

print("Preprocessing pipeline created successfully.")


# ============================================================
# 11. CREATE DECISION TREE CLASSIFIER
# ============================================================

# Create Decision Tree model
decision_tree = DecisionTreeClassifier(
    criterion="gini",
    max_depth=4,
    min_samples_leaf=10,
    random_state=42
)


# ============================================================
# 12. CREATE COMPLETE MODEL PIPELINE
# ============================================================

# Combine preprocessing and Decision Tree
model = Pipeline(
    steps=[
        ("preprocessor", preprocessor),
        ("classifier", decision_tree)
    ]
)


# ============================================================
# 13. TRAIN THE MODEL
# ============================================================

# Train model using training data
model.fit(X_train, y_train)

print("Decision Tree model trained successfully.")


# ============================================================
# 14. MAKE PREDICTIONS
# ============================================================

# Predict survival using testing data
y_pred = model.predict(X_test)

# Calculate prediction probabilities
y_probabilities = model.predict_proba(X_test)

# Extract survival probability (class 1)
survival_probability = y_probabilities[:, 1]


# ============================================================
# 15. CREATE PREDICTION RESULTS DATAFRAME
# ============================================================

# Compare actual and predicted values
results = pd.DataFrame({
    "Actual": y_test.values,
    "Predicted": y_pred,
    "Survival_Probability": survival_probability
})

print(results.head(10))


# ============================================================
# 16. ACTUAL SURVIVED VS PREDICTED SURVIVED
# ============================================================

# Select passengers who actually survived
actualServived = results[results["Actual"] == 1]

# Select correctly predicted survived passengers
predictedSurvived = actualServived[
    actualServived["Predicted"] == 1
]

# Display counts
print(
    f"Actual survived count: {len(actualServived)}"
    f"\nPredicted survived count: {len(predictedSurvived)}"
)

# Calculate correctly identified survived percentage
print(
    f"Predicted survived accuracy: "
    f"{len(predictedSurvived) / len(actualServived) * 100:.2f}%"
)


# ============================================================
# 17. ACTUAL NOT SURVIVED VS PREDICTED NOT SURVIVED
# ============================================================

# Select passengers who actually did not survive
actualNotServived = results[results["Actual"] == 0]

# Select correctly predicted not survived passengers
predictedNotSurvived = actualNotServived[
    actualNotServived["Predicted"] == 0
]

# Display actual not survived count
print(
    f"Actual not survived count: {len(actualNotServived)}"
)

# Display correctly predicted not survived count
print(
    f"Predicted not survived count: {len(predictedNotSurvived)}"
)

# Calculate correctly identified not survived percentage
print(
    f"Correctly identified not-survived percentage: "
    f"{len(predictedNotSurvived) / len(actualNotServived) * 100:.2f}%"
)


# ============================================================
# 18. FIND INCORRECT PREDICTIONS
# ============================================================

# Select rows where actual and predicted values are different
incorrectPred = results[
    results["Actual"] != results["Predicted"]
]

# Display incorrect predictions
print(incorrectPred)

# Count incorrect predictions
print("Total incorrect predictions:", len(incorrectPred))

# Count actual survived passengers incorrectly predicted
# as not survived
aspns = len(
    incorrectPred[incorrectPred["Actual"] == 1]
)

print("Actual survived but predicted not survived:", aspns)


# ============================================================
# 19. MODEL ACCURACY
# ============================================================

# Calculate overall accuracy
accuracy = accuracy_score(y_test, y_pred)

print("Accuracy:", round(accuracy * 100, 2), "%")


# ============================================================
# 20. CLASSIFICATION REPORT
# ============================================================

# Display precision, recall, F1-score and support
print(classification_report(
    y_test,
    y_pred,
    target_names=["Did not Survive", "Survived"],
    zero_division=0
))


# ============================================================
# 21. CONFUSION MATRIX
# ============================================================

# Display confusion matrix
ConfusionMatrixDisplay.from_predictions(
    y_test,
    y_pred,
    display_labels=[
        "Did Not Survive",
        "Survived"
    ],
    cmap="Blues"
)

plt.title("Decision Tree - Confusion Matrix")

plt.tight_layout()
plt.show()


# ============================================================
# 22. VISUALIZE DECISION TREE
# ============================================================

# Get feature names after preprocessing
feature_names = model.named_steps[
    "preprocessor"
].get_feature_names_out()

# Set figure size
plt.figure(figsize=(22, 12))

# Draw decision tree
plot_tree(
